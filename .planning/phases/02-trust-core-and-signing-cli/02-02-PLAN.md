- store.rs: trait Clock { fn now(&self) -> u64 }; trait SeqStore { fn get(&self, kind: DocKind) -> Option<(u64, [u8;32])>; fn put(&mut self, kind: DocKind, seq: u64, content_id: [u8;32]); }; trait ReplayCache { fn seen(&mut self, cmd_id: &[u8], expires_at: u64) -> bool /* true if already present; inserts otherwise */ }; enum DocKind { Manifest, Revocations, Techlist }; struct TrustStore { root_keys: Vec<PublicKey>, root_threshold: u8, manifest: Option<TrustManifest>, revocations: Option<ArtifactRevocations>, techlist: Option<TechnicianList> }
- TrustStore::new(compiled_roots: Vec<PublicKey>, threshold: u8); fn apply_manifest(&mut self, env_bytes: &[u8], clock: &dyn Clock, seq: &mut dyn SeqStore) -> Result<(), TrustError>; fn apply_revocations(..); fn apply_techlist(..); fn verify_role(&self, env: &Envelope, role: Role, now: u64) -> Result<Vec<KeyId>, TrustError> /* distinct valid signer key ids meeting threshold */; fn verify_command(&self, env_bytes: &[u8], agent_id: &str, client_id: &str, clock: &dyn Clock, replay: &mut dyn ReplayCache) -> Result<Command, TrustError>; fn trust_fresh(&self, now: u64) -> Result<(), TrustError> /* Stale if revocations missing or expired */; fn is_artifact_revoked(&self, content_id: &[u8;32]) -> bool
- TrustError adds: BelowThreshold { role, got, need }, WrongRole, KeyRevoked, CertNotValid, Expired, NotYetValid, Rollback { stored, got }, Equivocation, Stale, WrongAgent, WrongClient, Replay, NoManifest, ArtifactRevoked, InvalidDocument(String)
Edge/error cases:
- Manifest signed by a key that is a cert holder but not a root key -> BelowThreshold for root
- Two signatures by the same key -> counted once
- Threshold 0 in a manifest -> InvalidDocument (thresholds must be >= 1 and <= number of certs holding the role)
- Manifest that changes root_keys signed only by the old roots -> BelowThreshold
- Same seq, same payload -> Ok (idempotent); same seq, different payload -> Equivocation; lower seq -> Rollback
- Revocations expired by 1 s -> Stale; commands rejected; TrustUpdate still accepted
- Command for another agent_id or client_id, replayed cmd_id, expired, issued 301 s in the future -> each its own error
- Revoked signing key in manifest.revoked_keys signs a techlist -> KeyRevoked/BelowThreshold
